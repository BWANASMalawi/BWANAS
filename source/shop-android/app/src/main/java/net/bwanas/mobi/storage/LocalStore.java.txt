    public void completeAndQueueWithStockReservation(String operationId, String eventType, String payload,
                                                      String timestamp, boolean reserveStock, JSONObject fiscalInvoice) {
        if(net.bwanas.mobi.RequiredUpdatePolicy.blocked(state("required_app_version")))throw new IllegalStateException("Required app update must be installed first");
        SQLiteDatabase db = getWritableDatabase();
        db.beginTransaction();
        try {
            try (Cursor header=db.rawQuery("SELECT operation_type FROM operation_header WHERE operation_id=? AND status='draft'",new String[]{operationId})) {
                if(!header.moveToFirst())throw new IllegalStateException("Operation is not an open draft");
                if("sale".equals(header.getString(0)))payload=allocateReceipt(operationId,payload);
            }
            ContentValues completed = new ContentValues(); completed.put("payload",payload); completed.put("status", "completed"); completed.put("completed_at", timestamp);
            if (db.update("operation_header", completed, "operation_id=? AND status='draft'", new String[]{operationId}) != 1) throw new IllegalStateException("Operation is not an open draft");
            if (reserveStock) {
                validateSalePrices(db, payload);
                try (Cursor lines = db.rawQuery("SELECT sku,quantity FROM operation_line WHERE operation_id=? ORDER BY line_no", new String[]{operationId})) {
                    while (lines.moveToNext()) {
                        String sku = lines.getString(0); double quantity = lines.getDouble(1);
                        try (Cursor current = db.rawQuery("SELECT quantity FROM inventory WHERE sku=? AND active=1", new String[]{sku})) {
                            if (!current.moveToFirst() || current.getDouble(0) < quantity) throw new IllegalStateException("Insufficient local stock for " + sku);
                        }
                        db.execSQL("UPDATE inventory SET quantity=quantity+? WHERE sku=?", new Object[]{quantity * -1d, sku});
                    }
                }
            }
            ContentValues event = new ContentValues(); event.put("event_id", UUID.randomUUID().toString()); event.put("operation_id", operationId);
            event.put("event_type", eventType); event.put("payload", payload); event.put("status", "pending"); event.put("created_at", timestamp); event.put("updated_at", timestamp);
            db.insertOrThrow("outbox", null, event);
            if (fiscalInvoice != null) saveFiscalSale(db, operationId, fiscalInvoice, timestamp);
            db.setTransactionSuccessful();
        } finally { db.endTransaction(); }

    public static final class SalePriceChanged extends IllegalStateException {
        public SalePriceChanged(String sku) { super("Price changed for " + sku + ". Review the updated basket before confirming."); }

    private void validateSalePrices(SQLiteDatabase db, String payload) {
        try {
            JSONArray lines = new JSONObject(payload).optJSONArray("lines");
            if (lines == null) return;
            for (int i = 0; i < lines.length(); i++) {
                JSONObject line = lines.getJSONObject(i);
                String sku = line.getString("sku");
                try (Cursor current = db.rawQuery("SELECT i.unit_price,coalesce(d.state_value,'') FROM inventory i "
                        + "LEFT JOIN device_state d ON d.state_key='price_version_' || i.sku "
                        + "WHERE i.sku=? AND i.active=1", new String[]{sku})) {
                    if (!current.moveToFirst() || current.getDouble(0) <= 0
                            || Double.compare(current.getDouble(0), line.getDouble("unit_price")) != 0
                            || !current.getString(1).equals(line.optString("price_version_id", ""))) {
                        throw new SalePriceChanged(sku);
                    }
                }
            }
        } catch (JSONException error) {
            throw new IllegalArgumentException("Sale lines could not be checked", error);
        }
    }

    private void saveFiscalSale(SQLiteDatabase db, String operationId, JSONObject invoice, String timestamp) {
        if (hasUnsubmittedFiscalSales()) throw new IllegalStateException("A fiscal sale is already awaiting confirmation.");
        try {
            String prefix = (String) invoice.remove("numberPrefix");
            String counterKey = "mra_sequence_" + prefix;
            String previous = state(counterKey);
            long next = previous.isEmpty() ? 1 : Math.addExact(Long.parseLong(previous), 1);
            String invoiceNumber = prefix + "-" + net.bwanas.mobi.mra.MraInvoice.radix64(next);
            invoice.getJSONObject("invoiceHeader").put("invoiceNumber", invoiceNumber);
            saveState(counterKey, Long.toString(next));

            ContentValues row = new ContentValues();
            row.put("operation_id", operationId);
            row.put("fiscal_payload", invoice.toString());
            row.put("state", "ready");
            row.put("created_at", timestamp);
            row.put("updated_at", timestamp);
            db.insertOrThrow("mra_sale", null, row);
        } catch (JSONException error) {
            throw new IllegalStateException("The MRA invoice could not be saved.", error);
        }
    }
