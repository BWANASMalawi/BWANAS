    private void checkOfflineLimits(SQLiteDatabase db, JSONObject invoice, JSONObject limits) throws JSONException {
        java.math.BigDecimal amount = new java.math.BigDecimal(invoice.getJSONObject("invoiceSummary").get("invoiceTotal").toString());
        java.math.BigDecimal maxAmount = new java.math.BigDecimal(limits.get("maxCummulativeAmount").toString());
        double maxHours = limits.getDouble("maxTransactionAgeInHours");
        long profileTime = ((Number) invoice.remove("offlineProfileTime")).longValue();
        long now = ((Number) invoice.remove("offlineSaleTime")).longValue();
        if (maxHours <= 0 || maxAmount.signum() <= 0 || now < profileTime || now - profileTime >= maxHours * 3600000d) {
            throw new IllegalStateException("MRA offline time limit reached or offline sales are unavailable. Reconnect and sync.");
        }
        try (Cursor c = db.rawQuery(
                "SELECT fiscal_payload FROM mra_sale WHERE state IN ('offline_ready','offline_sending')", null)) {
            while (c.moveToNext()) {
                JSONObject queued = new JSONObject(c.getString(0));
                amount = amount.add(new java.math.BigDecimal(queued.getJSONObject("invoiceSummary").get("invoiceTotal").toString()));
            }
        }
        if (amount.compareTo(maxAmount) > 0) {
            throw new IllegalStateException("MRA offline sales amount limit reached. Reconnect and sync.");
        }
    }

    private void saveFiscalSale(SQLiteDatabase db, String operationId, JSONObject invoice, String timestamp) {
        JSONObject limits = (JSONObject) invoice.remove("offlineLimits");
        if (hasUnresolvedOnlineFiscalSale() || (limits == null && hasUnsubmittedFiscalSales())) {
            throw new IllegalStateException("A fiscal sale is already awaiting confirmation.");
        }
        try {
            if (limits != null) checkOfflineLimits(db, invoice, limits);
            String prefix = (String) invoice.remove("numberPrefix");
            String counterKey = "mra_sequence_" + prefix;
            String previous = state(counterKey);
            long next = previous.isEmpty() ? 1 : Math.addExact(Long.parseLong(previous), 1);
            String invoiceNumber = prefix + "-" + net.bwanas.mobi.mra.MraInvoice.radix64(next);
            invoice.getJSONObject("invoiceHeader").put("invoiceNumber", invoiceNumber);
            if (limits != null) {
                net.bwanas.mobi.mra.MraOfflineSignature.sign(invoice, prefix.substring(prefix.lastIndexOf('-') + 1),
                        net.bwanas.mobi.mra.MraActivationClient.offlineSigningKey(context));
            }
            saveState(counterKey, Long.toString(next));

            ContentValues row = new ContentValues();
            row.put("operation_id", operationId);
            row.put("fiscal_payload", invoice.toString());
            row.put("state", limits == null ? "ready" : "offline_ready");
            row.put("created_at", timestamp);
            row.put("updated_at", timestamp);
            db.insertOrThrow("mra_sale", null, row);
        } catch (IllegalStateException error) {
            throw error;
        } catch (Exception error) {
            throw new IllegalStateException("The MRA invoice could not be saved.", error);
        }
    }
